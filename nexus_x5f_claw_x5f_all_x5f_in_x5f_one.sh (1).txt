def ollama_chat(prompt, model="qwen2.5:3b"):
    try:
        r = requests.post(f"{OLLAMA_URL}/api/generate", json={"model":model,"prompt":prompt,"stream":False}, timeout=120)
        return r.json().get("response","")
    except Exception as e:
        return f"Ollama nicht erreichbar: {e} - starte mit 'ollama serve'"

@app.get("/api/state")
def get_state():
    try: state["cpu"] = int(psutil.cpu_percent())
    except: pass
    return state

@app.post("/api/start")
def api_start(bg: BackgroundTasks):
    state["running"]=True
    def loop():
        while state["running"]:
            for t in state["tests"]:
                if t["status"]=="WAIT":
                    t["status"]="RUN"
                    time.sleep(1)
                    t["status"]="OK"
                    state["logs"].append({"time":time.strftime("%H:%M:%S"),"type":"TEST","msg":f"Test {t['id']}/10 erfolgreich"})
                    break
            time.sleep(2)
    bg.add_task(loop)
    return {"ok":True}

@app.post("/api/pause")
def api_pause(): state["running"]=False; return {"ok":True}

@app.post("/api/stop")
def api_stop(): state["running"]=False; state["phase"]=1; return {"ok":True}

@app.post("/api/chat")
def api_chat(data: dict):
    p=data.get("prompt",""); m=data.get("model","qwen2.5:3b")
    return {"response": ollama_chat(p,m)}

@app.get("/")
def index(): return FileResponse(BASE / "web" / "index.html")

# 8. Start Script
cat > ~/nexus-claw/start.sh << 'SEOF'
#!/data/data/com.termux/files/usr/bin/bash
echo "Starte NEXUS CLAW..."
termux-wake-lock
ollama serve > ~/nexus-claw/logs/ollama.log 2>&1 &
sleep 2
cd ~/nexus-claw
python server.py &
echo "Server läuft: http://localhost:8000"
echo "Logs: tail -f ~/nexus-claw/logs/ollama.log"
echo "Agent in neuem Tab: python ~/nexus-claw/agent.py"

chmod +x ~/nexus-claw/server.py
