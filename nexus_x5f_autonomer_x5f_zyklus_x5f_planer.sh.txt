from fastapi import FastAPI, BackgroundTasks
from fastapi.responses import FileResponse
from fastapi.staticfiles import StaticFiles
import time, requests, json, pathlib, os
from pathlib import Path

app = FastAPI()
BASE = Path.home() / "nexus-claw"
OLLAMA = "http://localhost:11434"
MODEL_PLAN = "nexus-claw-planner"
MODEL_CODE = "qwen2.5-coder:1.5b"

state = {
    "modus": "IDLE",  # IDLE, PLANEN, PLAN_FERTIG, BAUEN, FERTIG
    "anzahl_zyklen": 0,
    "aktueller_plan_zyklus": 0,
    "aktueller_bau_zyklus": 0,
    "plaene": [],  # Liste von Zyklus Plänen
    "bau_fortschritt": [],
    "logs": [{"time":"18:36:00","type":"SYSTEM","msg":"NexusClaw bereit - Anzahl Zyklen eingeben"}],
    "autonom": True
}

def log(t, m):
    state["logs"].append({"time": time.strftime("%H:%M:%S"), "type": t, "msg": m})
    print(f"[{t}] {m}")

def ollama_gen(prompt, model=None):
    m = model or MODEL_PLAN
    try:
        r = requests.post(f"{OLLAMA}/api/generate", json={
            "model": m, "prompt": prompt, "stream": False,
            "options": {"temperature":0.2, "num_ctx":16384, "num_predict":2048}
        }, timeout=180)
        if r.status_code == 404 and m != MODEL_CODE:
            r = requests.post(f"{OLLAMA}/api/generate", json={
                "model": MODEL_CODE, "prompt": prompt, "stream": False
            }, timeout=180)
        return r.json().get("response","")
    except Exception as e:
        return f"FEHLER: {e}"

def plane_zyklus(zyklus_num, gesamt, vorherige_plaene):
    log("PLAN", f"Plane Zyklus {zyklus_num}/{gesamt}...")
    context = ""
    if vorherige_plaene:
        context = "Bisherige Zyklen:\n" + "\n".join([f"Z{p['zyklus']}: {p.get('ziel','')}" for p in vorherige_plaene])
    
    prompt = f"""
AUTONOMER PLANER - Zyklus {zyklus_num} von {gesamt} Zyklen

{context}

Aufgabe: Plane Zyklus {zyklus_num} für Unity Android Cyberpunk Racing Game MeinGame.

Zyklus {zyklus_num} Fokus Vorschlag:
- Zyklus 1: Core Loop + PlayerController + Basis Bewegung
- Zyklus 2: Gegner KI + A* Pfadberechnung
- Zyklus 3: Level System + Hindernisse
- Zyklus 4: Performance + Kollisionen + Object Pooling
- Zyklus 5+: Polishing + Menü + Sound + APK

Erstelle JSON (nur JSON, kein extra Text):
{{
  "zyklus": {zyklus_num},
  "ziel": "Klares Ziel für Zyklus {zyklus_num}",
  "aufgaben": [
    {{"id":"{zyklus_num}.1","titel":"PlayerController mit Touch","prio":"hoch","schaetzung":"2h"}},
    {{"id":"{zyklus_num}.2","titel":"...","prio":"mittel","schaetzung":"1h"}},
    {{"id":"{zyklus_num}.3","titel":"...","prio":"niedrig","schaetzung":"1h"}}
  ],
  "dod": "Definition of Done für Zyklus {zyklus_num}",
  "risiko": "Risiko + Mitigation"
}}

Nur JSON!
"""
    resp = ollama_gen(prompt, MODEL_PLAN)
    # Versuche JSON zu extrahieren
    try:
        # Finde JSON Block
        start = resp.find("{")
        end = resp.rfind("}")+1
        if start!=-1 and end!=-1:
            j = json.loads(resp[start:end])
        else:
            j = {"zyklus":zyklus_num,"ziel":f"Zyklus {zyklus_num} - Autonomer Bau","aufgaben":[{"id":f"{zyklus_num}.1","titel":resp[:50],"prio":"hoch"}],"dod":"Build erfolgreich","risiko":"kein"}
    except:
        j = {"zyklus":zyklus_num,"ziel":resp[:100],"aufgaben":[{"id":f"{zyklus_num}.1","titel":"Task aus Plan","prio":"hoch"}],"dod":"Done","risiko":""}
    
    # Speichern
    plan_file = BASE / f"plans/zyklus_{zyklus_num}.json"
    plan_file.parent.mkdir(exist_ok=True)
    plan_file.write_text(json.dumps(j, ensure_ascii=False, indent=2), encoding="utf-8")
    log("PLAN", f"Zyklus {zyklus_num} geplant: {j.get('ziel','')}")
    return j

def baue_zyklus(plan):
    z = plan["zyklus"]
    log("CODE", f"=== BAU Zyklus {z}: {plan.get('ziel','')} ===")
    for aufgabe in plan.get("aufgaben",[]):
        titel = aufgabe.get("titel","")
        log("CODE", f"Baue {aufgabe.get('id')}: {titel} [qwen2.5-coder:1.5b]")
        code_prompt = f"""
BAU Zyklus {z} Aufgabe {aufgabe.get('id')}: {titel}
Generiere Unity C# Script für Android.
Anforderungen:
- // File: Assets/Scripts/{titel.replace(' ','')[:20]}.cs als erste Zeile
- Touch Input, kein Keyboard
- Object Pooling, kein GC in Update
- Deutsch kommentiert
- Kompilierbar

Nur Code, kein Erklärungstext!
"""
        code = ollama_gen(code_prompt, MODEL_CODE)
        # Speichern
        scripts = BASE / "projects/MeinGame/Assets/Scripts"
        scripts.mkdir(parents=True, exist_ok=True)
        fname = f"Z{z}_{aufgabe.get('id','').replace('.','_')}_{titel.replace(' ','_')[:15]}.cs"
        (scripts / fname).write_text(code, encoding="utf-8")
        log("CODE", f" -> {fname} erstellt")
        time.sleep(1)
    log("TEST", f"Zyklus {z} Tests: Start, Kollision, Performance...")
    time.sleep(2)
    log("TEST", f"Zyklus {z} 3/3 Tests OK")

@app.get("/api/state")
def get_state():
    return state

@app.post("/api/plan_start")
def plan_start(data: dict, bg: BackgroundTasks):
    anzahl = int(data.get("anzahl_zyklen", 3))
    if anzahl <1 or anzahl>20:
        return {"error":"1-20 Zyklen erlaubt"}
    
    state["anzahl_zyklen"] = anzahl
    state["aktueller_plan_zyklus"] = 0
    state["aktueller_bau_zyklus"] = 0
    state["plaene"] = []
    state["bau_fortschritt"] = []
    state["modus"] = "PLANEN"
    log("SYSTEM", f"Autonome Planung gestartet: {anzahl} Zyklen")
    
    def autonomer_plan_loop():
        # PHASE 1: PLANEN
        for i in range(1, anzahl+1):
            state["aktueller_plan_zyklus"] = i
            plan = plane_zyklus(i, anzahl, state["plaene"])
            state["plaene"].append(plan)
            time.sleep(1)
        
        state["modus"] = "PLAN_FERTIG"
        log("SYSTEM", f"✅ Planung fertig: {anzahl} Zyklen geplant. Starte autonomen Bau in 3s...")
        time.sleep(3)
        
        # PHASE 2: AUTOMATISCH BAUEN - OHNE EINGRIFF
        state["modus"] = "BAUEN"
        log("SYSTEM", "🏗️ AUTONOMER BAU STARTET - Kein Eingriff nötig")
        for plan in state["plaene"]:
            state["aktueller_bau_zyklus"] = plan["zyklus"]
            baue_zyklus(plan)
        
        state["modus"] = "FERTIG"
        log("SYSTEM", f"🎉 ALLE {anzahl} ZYKLEN FERTIG! Projekt bereit in ~/nexus-claw/projects/MeinGame")
    
    bg.add_task(autonomer_plan_loop)
    return {"ok": True, "anzahl": anzahl, "modus": "PLANEN"}

@app.post("/api/stop")
def stop():
    state["modus"] = "IDLE"
    log("SYSTEM", "Gestoppt")
    return {"ok":True}

@app.get("/")
def idx():
    return FileResponse(BASE / "web" / "index.html")

app.mount("/web", StaticFiles(directory=str(BASE/"web")), name="web")

if __name__ == "__main__":
    import uvicorn
    print("NexusClaw Multi-Zyklus Planer auf http://localhost:8000")
    uvicorn.run(app, host="0.0.0.0", port=8000)
PYEOF
